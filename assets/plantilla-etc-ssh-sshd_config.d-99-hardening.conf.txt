# Plantilla de /etc/ssh/sshd_config.d/99-hardening.conf v1.0
# Hardening de SSH para servidor Debian 13
# Se requiere definir las variables MIPUERTO y MIUSUARIO

# --- Acceso ---
Port $MIPUERTO
PermitRootLogin no
PasswordAuthentication no
KbdInteractiveAuthentication no
ChallengeResponseAuthentication no
PubkeyAuthentication yes
UsePAM yes

# Limitar usuarios autorizados
AllowUsers $MIUSUARIO

# --- Endurecimiento ---
PermitEmptyPasswords no
MaxAuthTries 3
MaxSessions 5
LoginGraceTime 30
X11Forwarding no
AllowAgentForwarding no
AllowTcpForwarding no
PermitTunnel no

# --- Sesiones ---
ClientAliveInterval 300
ClientAliveCountMax 2

# --- Logging ---
LogLevel VERBOSE

# --- Criptografía moderna (ver 7.5) ---
KexAlgorithms curve25519-sha256,curve25519-sha256@libssh.org,sntrup761x25519-sha512@openssh.com
Ciphers chacha20-poly1305@openssh.com,aes256-gcm@openssh.com,aes128-gcm@openssh.com
MACs hmac-sha2-512-etm@openssh.com,hmac-sha2-256-etm@openssh.com
HostKeyAlgorithms ssh-ed25519,rsa-sha2-512,rsa-sha2-256
